**Example Implementation**:
```rust
async fn create_capability_lease(
    tool_name: &str,
    user_id: &str,
    thread_id: ThreadId,
    approval_context: &Option<ApprovalContext>,
) -> Result<CapabilityLease, Error> {
    // Check if tool requires approval
    let requires_approval = check_tool_approval_requirement(tool_name)?;
    
    if requires_approval {
        // Look up approval status
        let approval_status = lookup_approval_status(tool_name, user_id).await?;
        
        match approval_status {
            ApprovalStatus::Granted { actions, expires_at } => {
                // Create lease with restricted actions
                Ok(CapabilityLease {
                    id: LeaseId::new(),
                    thread_id,
                    capability_name: tool_name.to_string(),
                    granted_actions: GrantedActions::Specific(actions),
                    granted_at: Utc::now(),
                    expires_at,
                    max_uses: None,
                    uses_remaining: None,
                    revoked: false,
                    revoked_reason: None,
                })
            }
            ApprovalStatus::Pending => {
                Err(Error::ApprovalRequired {
                    tool_name: tool_name.to_string(),
                })
            }
            ApprovalStatus::Denied { reason } => {
                Err(Error::ApprovalDenied {
                    tool_name: tool_name.to_string(),
                    reason,
                })
            }
        }
    } else {
        // Tool doesn't require approval - grant all actions
        Ok(CapabilityLease {
            id: LeaseId::new(),
            thread_id,
            capability_name: tool_name.to_string(),
            granted_actions: GrantedActions::All,
            granted_at: Utc::now(),
            expires_at: None,
            max_uses: None,
            uses_remaining: None,
            revoked: false,
            revoked_reason: None,
        })
    }
}
```

### Task 3: Implement Real Gate Controller

**Current State**: Using `CancellingGateController` which denies all gates

**Actions**:
1. Create `ApprovalGateController` that integrates with approval system
2. Implement gate pause/resume logic for approval requests
3. Handle approval notifications to user
4. Implement approval response handling
5. Support inline retry after approval granted

**Example Implementation**:
```rust
struct ApprovalGateController {
    approval_manager: Arc<ApprovalManager>,
    notification_sender: Arc<dyn NotificationSender>,
}

#[async_trait::async_trait]
impl GateController for ApprovalGateController {
    async fn handle_gate(
        &self,
        gate_name: &str,
        action_name: &str,
        call_id: &str,
        parameters: &serde_json::Value,
        context: &GateContext,
    ) -> GateDecision {
        match gate_name {
            "approval" => {
                // Check if approval already granted for this call
                if let Some(approval) = self.approval_manager
                    .get_approval(call_id)
                    .await
                {
                    return GateDecision::Allow;
                }
                
                // Request approval from user
                let request = ApprovalRequest {
                    call_id: call_id.to_string(),
                    action_name: action_name.to_string(),
                    parameters: parameters.clone(),
                    user_id: context.user_id.clone(),
                    conversation_id: context.conversation_id,
                };
                
                self.approval_manager.request_approval(request).await?;
                self.notification_sender.send_approval_request(&request).await?;
                
                // Pause execution until approval received
                GateDecision::Pause {
                    resume_kind: ResumeKind::Approval,
                }
            }
            _ => GateDecision::Deny {
                reason: format!("Unknown gate: {}", gate_name),
            },
        }
    }
}
```

- Cache approval status to avoid repeated lookups
- Async approval handling to avoid blocking
- Efficient notification delivery

- Support tools that don't require approval
- Graceful degradation if approval system unavailable
- Maintain existing error handling behavior
